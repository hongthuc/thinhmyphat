        /* Điều khiển Slider 2 Banner tự trượt */
        let currentBannerIdx = 0;
        let bannerInterval = null;

        function showBannerSlide(idx) {
            const slides = document.querySelectorAll('.banner-slide');
            const dots = document.querySelectorAll('.banner-dot');
            currentBannerIdx = (idx + slides.length) % slides.length;
            slides.forEach((sl, i) => {
                if (i === currentBannerIdx) {
                    sl.classList.remove('hidden');
                    sl.classList.add('block');
                } else {
                    sl.classList.remove('block');
                    sl.classList.add('hidden');
                }
            });
            dots.forEach((dot, i) => {
                if (i === currentBannerIdx) {
                    dot.className = "banner-dot w-6 h-1.5 rounded-full bg-gold-400 transition-all";
                } else {
                    dot.className = "banner-dot w-2.5 h-1.5 rounded-full bg-white/50 transition-all";
                }
            });
        }

        function changeBannerSlide(step) {
            showBannerSlide(currentBannerIdx + step);
            resetBannerAutoSlide();
        }

        function goToBannerSlide(idx) {
            showBannerSlide(idx);
            resetBannerAutoSlide();
        }

        function resetBannerAutoSlide() {
            if (bannerInterval) clearInterval(bannerInterval);
            bannerInterval = setInterval(() => {
                showBannerSlide(currentBannerIdx + 1);
            }, 5000);
        }

        /* Đóng/Mở Menu Điện thoại */
        function toggleMobileMenu() {
            const menu = document.getElementById('mobileMenuDropdown');
            const icon = document.getElementById('mobileMenuIcon');
            if (menu.classList.contains('hidden')) {
                menu.classList.remove('hidden');
                icon.className = 'fa-solid fa-xmark text-base';
            } else {
                menu.classList.add('hidden');
                icon.className = 'fa-solid fa-bars text-base';
            }
        }

        /* Tìm kiếm nhanh sản phẩm */
        function filterProductsByQuery(queryText) {
            const q = (queryText || '').trim().toLowerCase();
            const cards = document.querySelectorAll('[data-product-name]');
            cards.forEach(card => {
                const name = card.getAttribute('data-product-name');
                if (!q || name.includes(q)) {
                    card.style.display = '';
                } else {
                    card.style.display = 'none';
                }
            });
            const firstSection = document.getElementById('dm-cau-thang-go');
            if (firstSection && q) {
                firstSection.scrollIntoView({ behavior: 'smooth' });
            }
        }

        function filterProductsBySearch() {
            const val = document.getElementById('headerSearchInput').value;
            filterProductsByQuery(val);
        }

        function filterProductsByMobileSearch() {
            const val = document.getElementById('mobileSearchInput').value;
            toggleMobileMenu();
            filterProductsByQuery(val);
        }

        /* Mở / Đóng Modal chi tiết sản phẩm */
        function openProductModal(title, cat, imgUrl, desc, driveId, fallback) {
            document.getElementById('modalTitle').textContent = title;
            document.getElementById('modalCategory').textContent = "Danh mục: " + cat;
            document.getElementById('modalDesc').textContent = desc;
            const modalImg = document.getElementById('modalImg');
            modalImg.setAttribute('data-drive-id', driveId || '');
            modalImg.removeAttribute('data-tried-lh3');
            modalImg.onerror = function() {
                handleDriveImageError(modalImg, fallback);
            };
            modalImg.src = imgUrl;
            document.getElementById('productModal').classList.remove('hidden');
        }
